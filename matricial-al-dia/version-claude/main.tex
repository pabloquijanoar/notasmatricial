\begin{pb}

Determinar los valores de $L$ para los que el siguiente problema tiene (al menos) una solución no trivial:
\begin{equation*}
\begin{aligned}
y''(x) + y(x) &= 0, \qquad & &0 \le x \le L, \\
y(0) &= 0 , \\
y(L) &= 0.
\end{aligned}
\end{equation*}
Para dichos valores de $L$, ¿cuántas soluciones tiene?

\end{pb}

\begin{pbstar}
 Sea $y(x)$ una autofunción del problema de Sturm-Liouville
\begin{equation*}\tag{SL}
\begin{aligned}
\frac{d}{dx} \left[ K(x) \frac{dy}{dx} \right]
+ \left[ q(x) + \lambda g(x) \right] y &= 0
\qquad & &a \le x \le b, \\
c_1 \, y(a) + c_2 \, y'(a) &= 0 \qquad&  &c_1^2 + c_2^2 \neq 0 \\
c_3 \, y(b) + c_4 \, y'(b) &= 0 \qquad&  &c_3^2 + c_4^2 \neq 0
\end{aligned}
\end{equation*}
(suponiendo que $K'$, $g$, $q$ son continuas en $[a,b]$ con $K(x) > 0$ para todo $x \in [a,b]$).

Demostrar que todos los ceros de $y(x)$ en $[a,b]$ son simples.

\end{pbstar}

\begin{pbstar}
 Sea $y(x)$ una solución no-trivial de la ecuación de Sturm-Liouville
\begin{equation*}
\frac{d}{dx} \left[ K(x) \frac{dy}{dx} \right]
+ \left[ q(x) + \lambda g(x) \right] y = 0
\qquad\qquad a \le x \le b,
\end{equation*}
(suponiendo que $K'$, $g$, $q$ son continuas en $[a,b]$ con $K(x) > 0$ para todo $x \in [a,b]$).

Demostrar que $y(x)$ tiene a lo sumo un número finito de ceros en $[a,b]$.
\end{pbstar}
